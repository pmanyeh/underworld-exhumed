"""Run the failed upstream settings slice with Windows-valid temporary folder names."""

import sys

workspace = Path(__file__).absolute().parent.parent
sys.path.insert(0, str(workspace / "underworld-exhumed/exhume/tools"))

original_fresh = setcheck.fresh

setcheck.fresh = fresh
