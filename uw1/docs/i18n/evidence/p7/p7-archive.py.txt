"""Archive P7 evidence without distributing private game/runtime assets."""
from pathlib import Path
import hashlib
import json

workspace = Path(__file__).resolve().parent.parent
repo = workspace / 'underworld-exhumed'
uw = repo / 'uw1'
out = uw / 'docs/i18n/evidence/p7'
out.mkdir(parents=True, exist_ok=True)

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def copy_text(src, name):
    data = src.read_bytes().replace(b'\r\n', b'\n').rstrip(b'\n') + b'\n'
    (out / name).write_bytes(data)

for src in (uw / 'build/i18n').glob('p7-*'):
    if src.is_file() and src.suffix in ('.json', '.log'):
        copy_text(src, src.name.removeprefix('p7-'))

game = uw / 'build/hd-scroll/game'
for name in ('game-tests.json', 'modern-replay.json'):
    copy_text(game / name, name)
cases = json.loads((game / 'game-tests.json').read_text(encoding='utf-8'))['cases']
for name in cases:
    for ext in ('.log', '.jsonl'):
        copy_text(game / (name + ext), name + ext)
for src in (uw / 'build/hd-scroll/cli').glob('*'):
    if src.suffix in ('.log', '.json'):
        copy_text(src, src.name if src.name.startswith('cli-') else 'cli-' + src.name)

for name in ('p7-game-tests.py', 'p7-modern-replay.py', 'p7-replay-en.py',
             'p7-native-regression.py', 'prepare-fuzzhost-link.py', 'p7-archive.py'):
    copy_text(workspace / '.poc-tools' / name, name + '.txt')

images = {}
def copy_image(src, name):
    with Image.open(src) as image:
        original = image.tobytes()
        mode, size = image.mode, image.size
        image.save(out / name, optimize=True, compress_level=9)
    with Image.open(out / name) as image:
        assert image.mode == mode and image.size == size and image.tobytes() == original
    images[name] = {'sha256': sha(out / name), 'decoded_pixel_sha256': hashlib.sha256(original).hexdigest(),
                    'mode': mode, 'size': list(size), 'lossless_recompression_verified': True}

for name in cases:
    copy_image(game / (name + '.png'), name + '.png')
    copy_image(game / (name + '-scanout.png'), name + '-scanout.png')
copy_image(uw / 'build/hd-scroll/window-asan-3.png', 'resize.png')
